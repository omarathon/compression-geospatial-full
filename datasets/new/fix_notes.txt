worldclim temperatures were quantized by multiplying by 10 (it's only accurate to .1 deg so its fine)

for region in nyc iowa himalayas sahara amazon; do     gdal_translate -ot Int16 -scale 0 1 0 10 -a_nodata -32768         WorldClim_temp_${region}.tif WorldClim_temp_${region}_int16.tif;     mv WorldClim_temp_${region}_int16.tif WorldClim_temp_${region}.tif; done

etopo1 high-res is float32 so it was quantized to int32 by multipliying by 1000:

for region in nyc iowa himalayas sahara amazon ocean; do
    gdal_calc.py \
        -A etopo_highres/ETOPO_HighRes_${region}.tif \
        --calc="A*1000" \
        --outfile=etopo_highres_quant/ETOPO_HighRes_${region}.tif \
        --type=Int32 \
        --NoDataValue=-2147483647 \
        --overwrite
done
